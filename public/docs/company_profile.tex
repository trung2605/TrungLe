\section{Kiến Trúc Kỹ Thuật \& Dòng Chảy Tự Động Hóa SaaS (Closed-Loop Automation)}

\begin{tcolorbox}[colback=tagbg,colframe=accent,arc=2.5mm,boxrule=0.8pt,top=1.5mm,bottom=1.5mm]
\small
\textbf{\color{accent} SỐ LIỆU THÚC ĐẨY QUYẾT ĐỊNH:} Theo dữ liệu từ Ngân hàng Nhà nước Việt Nam (NHNN), thanh toán QR trong 9 tháng đầu năm 2025 \textbf{tăng 61,63\% về lượng giao dịch và 150,67\% về giá trị} so với cùng kỳ. Gần \textbf{90 triệu tài khoản Mobile Banking} đang quét VietQR tại hơn \textbf{2,1 triệu điểm chấp nhận} toàn quốc. Với quy mô này, việc không tích hợp VietQR tự động đồng nghĩa với việc mất đi cơ hội chuyển đổi ngay tại điểm quyết định mua hàng.
\end{tcolorbox}

\vspace{1.5mm}

Một website bán hàng hiện đại không chỉ dừng lại ở trang giới thiệu tĩnh. Để tạo doanh thu bền vững và cắt giảm tối đa chi phí nhân sự vận hành, hệ thống được thiết kế theo mô hình \textbf{Tự Động Hóa Khép Kín (Closed-Loop Automation)}:

\begin{figure}[h!]
    \centering
    \includegraphics[width=0.97\textwidth,height=7.5cm,keepaspectratio]{images/architecture_infographic.png}
    \caption*{\footnotesize Hình 1: Sơ đồ luồng dữ liệu SaaS Automation --- Kết nối liền mạch: Khách hàng tương tác $\rightarrow$ Nhận diện đơn hàng $\rightarrow$ Khớp lệnh VietQR tức thì $<0.5\text{s}$ $\rightarrow$ Đồng bộ Kho vận $\rightarrow$ AI Agent chăm sóc 24/7}
\end{figure}

\subsection{Bốn Trụ Cột Kỹ Thuật Nền Tảng}

\begin{enumerate}[leftmargin=*,itemsep=3pt,topsep=1pt]
    \item \textbf{VietQR Động \& Webhooks Tức Thì (Thời gian xử lý $<0.5$ giây):}\\
    Khách hàng chọn mua $\rightarrow$ hệ thống tự sinh mã VietQR chứa đúng số tiền và mã đơn duy nhất. Webhook ngân hàng bắn callback về server, hệ thống tự động kiểm tra chữ ký số, gạch đơn, gửi hóa đơn điện tử qua Email và thông báo Zalo/Telegram cho chủ shop---tất cả trong chưa đầy nửa giây, không cần nhân viên đối soát thủ công. \textbf{Tiết kiệm 100\% chi phí nhân sự kiểm tra chuyển khoản hằng ngày.}

    \item \textbf{AI Agent DeepSeek --- Nhân viên Kinh Doanh Ảo Hoạt Động 24/7:}\\
    Theo nghiên cứu của VnEconomy (2025), \textbf{93\% doanh nghiệp Việt Nam} đã tích hợp công cụ AI và \textbf{86\% SME ghi nhận tăng trưởng doanh thu} sau khi ứng dụng AI tư vấn cá nhân hóa. AI Agent của chúng tôi được xây dựng trên nền tảng DeepSeek---chi phí API chỉ \textbf{\$0.30--\$1.20/triệu token} (so với GPT-5.5 ở mức $>$\$5/triệu token), tức là \textbf{tiết kiệm 80--95\% chi phí AI} mà vẫn đạt hiệu suất tư vấn vượt trội trong lĩnh vực sản phẩm và bán hàng. Agent tự học danh mục sản phẩm, tư vấn đặc điểm kỹ thuật, gợi ý bán kèm (Cross-sell/Up-sell) và dẫn dắt khách đến hành vi thanh toán ngay trong cửa sổ chat.

    \item \textbf{Hiệu Năng Google Core Web Vitals --- Nền Tảng SEO Bền Vững:}\\
    Theo Google (2024), trải nghiệm người dùng được đo bằng 03 chỉ số bắt buộc: LCP $\leq 2.5\text{s}$ (tốc độ tải nội dung chính), INP $\leq 200\text{ms}$ (độ phản hồi tương tác), CLS $\leq 0.1$ (ổn định giao diện). Website của chúng tôi xây dựng theo kiến trúc tối ưu nén ảnh WebP/AVIF, Lazy Loading thông minh và CSS purging---điểm Lighthouse duy trì \textbf{95--100/100}, đảm bảo xếp hạng tốt trên Google tìm kiếm tự nhiên và giảm tỷ lệ thoát trang (bounce rate) xuống thấp nhất.

    \item \textbf{Tự Động Hóa Quy Trình Bằng n8n Workflow Engine:}\\
    n8n---nền tảng tự động hóa đang được \textbf{15.000+ doanh nghiệp} trên toàn cầu (bao gồm 3.000+ khách hàng doanh nghiệp) sử dụng, với định giá thị trường \textbf{\$5.2 tỷ USD} (2026)---cho phép kết nối linh hoạt hàng trăm nguồn dữ liệu: Email, Zalo, Google Sheets, CRM, kho hàng, ngân hàng. Trên \textbf{80\% luồng tự động hóa} hiện tại trên n8n đã tích hợp AI Agent, tạo ra hệ sinh thái tự vận hành mà không cần phần mềm ERP đắt tiền.
\end{enumerate}

Dự án nghiên cứu chuyên sâu kết hợp Cây Cú Pháp Trừu Tượng (Abstract Syntax Tree --- AST) với kiến trúc RAG (Retrieval-Augmented Generation) để phát hiện và tự động gợi ý sửa lỗi xung đột tài nguyên đa luồng (Race Condition) trong môi trường tải cao:

\begin{center}
    \includegraphics[width=0.90\textwidth,height=6.0cm,keepaspectratio]{images/threadlearn_pipeline.png}\\[1mm]
    \captionof{figure}{\footnotesize Kiến trúc Pipeline ThreadLearn: Phân tích AST $\rightarrow$ Phát hiện Race Condition $\rightarrow$ RAG Retrieval $\rightarrow$ LLM Engine tạo gợi ý sửa lỗi thời gian thực}
\end{center}

\textbf{Giá trị ứng dụng thực tiễn cho khách hàng:} Năng lực phân tích thuật toán đa luồng chuyên sâu giúp đội ngũ chúng tôi thiết kế và xây dựng những hệ thống xử lý giao dịch đồng thời (High Concurrency Transaction Systems) cho các sàn thương mại và website bán hàng---loại bỏ triệt để nguy cơ trừ tiền hai lần hoặc bán trùng một sản phẩm cho nhiều khách cùng một thời điểm. \section{Quy Trình Triển Khai, Điều Khoản Hợp Đồng \& Cam Kết Pháp Lý}

\begin{enumerate}[leftmargin=*,itemsep=3pt]
    \item \textbf{Quy trình giải ngân bảo vệ an toàn 40\% / 60\%:}
    \begin{itemize}[itemsep=1.5pt]
        \item \textit{Đợt 1 --- 40\% (Cọc kỹ thuật):} Thanh toán sau khi hai bên ký tắt Tài liệu Đặc tả Yêu cầu (SRS) và khách hàng phê duyệt bản Wireframe/Prototype giao diện trên Figma.
        \item \textit{Đợt 2 --- 60\% (Nghiệm thu):} Chỉ thanh toán sau khi hệ thống hoàn thiện 100\%, kiểm thử nghiệm thu không còn lỗi nghiêm trọng (zero critical bug) và hệ thống đã chạy thử nghiệm thành công trên tên miền chính thức của khách hàng trong ít nhất 48 giờ.
    \end{itemize}

    \item \textbf{Cam kết tiến độ Thưởng / Phạt minh bạch:}
    \begin{itemize}[itemsep=1.5pt]
        \item \textit{Thời gian ân hạn (Grace Period):} Tối đa \textbf{02 ngày làm việc} cho các trường hợp bất khả kháng được ghi nhận trong biên bản.
        \item \textit{Phạt tiến độ (vi phạm từ phía đội phát triển):} \textbf{1\% giá trị hợp đồng / ngày trễ}, tối đa 10\%. Trễ quá \textbf{15 ngày}: \textbf{hoàn trả 100\% tiền cọc ngay lập tức, không tranh chấp}.
        \item \textit{Thưởng bàn giao sớm:} Nếu hoàn thành trước tiến độ và chất lượng vượt kỳ vọng, khách hàng có thể cân nhắc thưởng thêm theo thỏa thuận.
    \end{itemize}

    \item \textbf{Chính sách xử lý tính năng phát sinh (Feature Request):}
    \begin{itemize}[itemsep=1.5pt]
        \item \textit{Hoán đổi tính năng tương đương --- Miễn phí:} Trong thời gian phát triển, khách hàng được quyền hoán đổi tối đa \textbf{20\%} khối lượng tính năng sang tính năng tương đương về độ phức tạp, mà không mất thêm chi phí.
        \item \textit{Nâng cấp Live System sau nghiệm thu (+20\%):} Nếu muốn nâng cấp gói MVP lên gói Toàn Diện sau khi hệ thống đã chạy thật, áp dụng phụ phí 20\% cho công tác refactor mã nguồn và kiểm thử hồi quy (Regression Testing) không làm ảnh hưởng người dùng đang sử dụng.
    \end{itemize}

    \item \textbf{Bảo hành \& Hỗ trợ kỹ thuật dài hạn (SLA):}
    \begin{itemize}[itemsep=1.5pt]
        \item Bảo hành lỗi kỹ thuật từ \textbf{6--12 tháng} theo gói dịch vụ đã ký.
        \item \textbf{Hỗ trợ khẩn cấp 24/7} đối với lỗi nghiêm trọng ảnh hưởng đến giao dịch (Gói 3). Thời gian phản hồi tối đa: 2 giờ đối với sự cố P1 (Production Down).
        \item Hỗ trợ cập nhật bảo mật định kỳ (Security Patch) và tư vấn nâng cấp kỹ thuật sau khi hết thời hạn bảo hành.
    \end{itemize}

    \item \textbf{Nghĩa vụ Pháp lý \& Bảo mật Thông tin:}\\
    Đội ngũ chúng tôi tự chịu trách nhiệm kê khai và nộp thuế thu nhập cá nhân (TNCN) theo đúng quy định pháp luật Việt Nam hiện hành. Hợp đồng dịch vụ bao gồm điều khoản \textbf{Bảo mật thông tin kinh doanh (NDA)} và cam kết \textbf{chuyển giao toàn bộ bản quyền sở hữu trí tuệ} cho khách hàng sau khi nghiệm thu và thanh toán đợt 2.
\end{enumerate}
